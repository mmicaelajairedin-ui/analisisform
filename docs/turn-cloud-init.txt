#!/bin/bash
# ── Auto-instalación de coturn (TURN) — pegar en el campo "Cloud config" de Hetzner
# al CREAR el servidor. El servidor se instala y configura SOLO, sin terminal.
set -e
apt-get update
apt-get install -y coturn curl
EXTIP=$(curl -s https://api.ipify.org || hostname -I | awk '{print $1}')
cat > /etc/turnserver.conf <<EOF
listening-port=3478
fingerprint
lt-cred-mech
user=pathway:PathwayTurn2026xk9q
realm=pathwaycareercoach.com
external-ip=$EXTIP
min-port=49152
max-port=65535
no-cli
EOF
echo 'TURNSERVER_ENABLED=1' > /etc/default/coturn
systemctl enable coturn
systemctl restart coturn
